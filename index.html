📡 Bloque 4 — Telemetría Silenciosa en Segundo Plano
💡 ¿Qué hace este bloque y por qué este código?
Define la clase TelemetryLogger para auditar de forma silenciosa cada interacción en telemetria_uso_agente.csv. Registra timestamp, rol, caracteres intercambiados y si se generó un informe HTML. Esto permite medir el uso real del prototipo en la organización.
Celda de Código 4: Telemetría en Background Python (Colab) 
import csv
from datetime import datetime
import uuid

class TelemetryLogger:
    """Registrador de telemetría y auditoría en segundo plano."""
    def __init__(self, filepath="telemetria_uso_agente.csv"):
        self.filepath = filepath
        self.session_id = str(uuid.uuid4())[:8]
        self._init_file()

    def _init_file(self):
        if not os.path.exists(self.filepath):
            with open(self.filepath, "w", newline="", encoding="utf-8") as f:
                writer = csv.writer(f)
                writer.writerow(["timestamp", "session_id", "actor", "prompt_len", "resp_len", "generated_report", "categoria"])

    def log(self, actor, prompt_len, resp_len, generated_report=False, categoria="dialogo"):
        try:
            with open(self.filepath, "a", newline="", encoding="utf-8") as f:
                writer = csv.writer(f)
                writer.writerow([
                    datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                    self.session_id, actor, prompt_len, resp_len,
                    "SI" if generated_report else "NO", categoria
                ])
        except Exception:
            pass

    def get_summary_df(self):
        return pd.read_csv(self.filepath) if os.path.exists(self.filepath) else pd.DataFrame()

telemetry = TelemetryLogger()
print("✔ Telemetría activa en segundo plano. Sesión iniciada.")

🚀 Bloque 6 — Interfaz Web con Gradio 6, Subida de Archivos y Descarga Directa
💡 Mejora Clave en Gradio: Adaptado al formato nativo de Gradio 6 (diccionarios [{{"role": "user", "content": ...}}]) e incorpora el componente gr.File para que el usuario pueda arrastrar y soltar cualquier CSV o Excel. Pandas lee el archivo, extrae columnas e inyecta dinámicamente los datos en el System Prompt.
Celda de Código 6: Interfaz Gradio con Soporte de Carga de Archivos Python (Colab) 
import gradio as gr
import re
import pandas as pd
from langchain_core.messages import SystemMessage, HumanMessage, AIMessage

chat_history = [SystemMessage(content=SYSTEM_PROMPT)]
latest_html_file = "reporte_ejecutivo_icesi.html"

def extract_html_code(text):
    match = re.search(r"```html\s*(.*?)\s*```", text, re.DOTALL)
    return match.group(1).strip() if match else None

def responder_agente(mensaje_usuario, archivo_adjunto, historial_ui):
    global chat_history, latest_html_file

    if not mensaje_usuario.strip() and archivo_adjunto:
        mensaje_usuario = "He cargado un nuevo archivo de datos para analizar."

    if not mensaje_usuario.strip() and not archivo_adjunto:
        return "", historial_ui, gr.update(), gr.update(), telemetry.get_summary_df()

    # Procesar archivo CSV o Excel adjunto por el usuario
    if archivo_adjunto is not None:
        try:
            filepath = archivo_adjunto.name
            if filepath.endswith('.csv'):
                try:
                    df_usuario = pd.read_csv(filepath, encoding='utf-8')
                except UnicodeDecodeError:
                    df_usuario = pd.read_csv(filepath, encoding='latin-1')
            else:
                df_usuario = pd.read_excel(filepath)

            resumen_datos = df_usuario.head(15).to_csv(index=False)
            info_columnas = ", ".join(df_usuario.columns)
            prompt_dinamico = SYSTEM_PROMPT.replace(dataset_summary, f"--- DATOS CARGADOS POR EL USUARIO ---\nColumnas: {info_columnas}\nMuestra:\n{resumen_datos}")
            chat_history[0] = SystemMessage(content=prompt_dinamico)
            mensaje_usuario += f" (Adjunto archivo con {len(df_usuario)} filas y columnas: {info_columnas})"
        except Exception as e:
            mensaje_usuario += f" (Error al procesar archivo: {str(e)})"

    chat_history.append(HumanMessage(content=mensaje_usuario))
    respuesta = llm.invoke(chat_history)
    texto_respuesta = respuesta.content
    chat_history.append(AIMessage(content=texto_respuesta))

    # Formato compatible con Gradio 6
    historial_ui.append({"role": "user", "content": mensaje_usuario})
    historial_ui.append({"role": "assistant", "content": texto_respuesta})

    html_code = extract_html_code(texto_respuesta)
    archivo_descarga = None
    preview_html = gr.update()

    if html_code:
        with open(latest_html_file, "w", encoding="utf-8") as f:
            f.write(html_code)
        archivo_descarga = latest_html_file
        preview_html = gr.HTML(value=f'<iframe srcdoc="{html_code.replace(""", """)}" style="width:100%; height:450px; border:1px solid #cbd5e1; border-radius:8px;"></iframe>')
        telemetry.log("Agente", len(mensaje_usuario), len(texto_respuesta), generated_report=True, categoria="generacion_html")
    else:
        telemetry.log("Agente", len(mensaje_usuario), len(texto_respuesta), generated_report=False, categoria="planificacion")

    return "", historial_ui, preview_html, (archivo_descarga if html_code else gr.update()), telemetry.get_summary_df()
